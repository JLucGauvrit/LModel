 44    def encode(self, x: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
 45        """
 46        Encode une observation en paramètres de la distribution latente q(z|x).
 47
 48        :param x: Image normalisée dans [0, 1], shape (B, 3, 7, 7).
 49        :type x: torch.Tensor

 57    def reparameterize(self, mu: torch.Tensor, log_var: torch.Tensor) -> torch.Tensor:
 58        """
 59        Trick de reparamétrisation : z = μ + ε·σ avec ε ~ N(0, I).
 60
 61        Permet la rétropropagation à travers l'échantillonnage stochastique.
 62        En mode ``eval()``, retourne directement μ (mode de la distribution).
 63
 64        :param mu: Moyenne de la distribution latente, shape (B, latent_dim).
 65        :type mu: torch.Tensor
 66        :param log_var: Log-variance, shape (B, latent_dim).
 67        :type log_var: torch.Tensor
 68        :returns: Vecteur latent z, shape (B, latent_dim).
 69        :rtype: torch.Tensor
 70        """
 71        if self.training:
 72            std = torch.exp(0.5 * log_var)
 73            return mu + torch.randn_like(std) * std
 74        return mu

 76    def decode(self, z: torch.Tensor) -> torch.Tensor:
 77        """
 78        Décode un vecteur latent en reconstruction de l'observation.

110        Loss ELBO = reconstruction MSE + β·KL divergence.

115        :type x_recon: torch.Tensor
116        :param mu: Moyenne latente, shape (B, latent_dim).
117        :type mu: torch.Tensor
118        :param log_var: Log-variance latente, shape (B, latent_dim).
119        :type log_var: torch.Tensor
120        :param beta: Coefficient KL (β-VAE). 1.0 = VAE standard.
121        :type beta: float
122        :returns: Loss scalaire.
123        :rtype: torch.Tensor
124        """
125        recon = F.mse_loss(x_recon, x, reduction='sum') / x.size(0)
126        kl = -0.5 * torch.mean(1 + log_var - mu.pow(2) - log_var.exp())

 12
 13    :param latent_dim: Dimension du vecteur latent z (sortie du VAE).
 14    :type latent_dim: int
 15    :param action_dim: Nombre d'actions (7 pour MiniGrid), encodées en one-hot.
 16    :type action_dim: int
 17    :param hidden_dim: Dimension de l'état caché LSTM.
 18    :type hidden_dim: int
 19    :param num_gaussians: Nombre de composantes dans le mélange de gaussiennes.
 20    :type num_gaussians: int
 21    """
 22
 23    def __init__(
 24        self,
 25        latent_dim: int = 32,
 26        action_dim: int = 7,
 27        hidden_dim: int = 256,
 28        num_gaussians: int = 5,
 29    ):
 30        super().__init__()
 31        self.latent_dim = latent_dim
 32        self.action_dim = action_dim

 44        self,
 45        z: torch.Tensor,
 46        action: torch.Tensor,
 47        h: torch.Tensor,
 48        c: torch.Tensor,
 49    ) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
 50        """
 51        Un pas de temps : met à jour l'état caché et prédit la distribution de z_{t+1}.
 52
 53        :param z: Vecteur latent courant, shape (B, latent_dim).
 54        :type z: torch.Tensor
 55        :param action: Indice d'action scalaire, shape (B,).
 56        :type action: torch.Tensor
 57        :param h: État caché LSTM précédent, shape (B, hidden_dim).
 58        :type h: torch.Tensor
 59        :param c: État cellule LSTM précédent, shape (B, hidden_dim).
 60        :type c: torch.Tensor
 61        :returns: Tuple (pi, mu, sigma, h_new, c_new).
 62
 63                  - pi : logits de mélange (B, num_gaussians)
 64                  - mu : moyennes (B, num_gaussians, latent_dim)
 65                  - sigma : écarts-types > 0 (B, num_gaussians, latent_dim)
 66                  - h_new, c_new : nouvel état caché LSTM
 67        :rtype: tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]
 68        """
 69        a_onehot = torch.zeros(z.size(0), self.action_dim, device=z.device)
 70        a_onehot.scatter_(1, action.view(-1, 1).long(), 1.0)
 71
 72        h_new, c_new = self.lstm(torch.cat([z, a_onehot], dim=1), (h, c))
 73
 74        out = self.mdn_head(h_new)
 75        K, D = self.num_gaussians, self.latent_dim
 76
 77        pi        = out[:, :K]
 78        mu        = out[:, K : K * (1 + D)].view(-1, K, D)
 79        log_sigma = out[:, K * (1 + D) :].view(-1, K, D)
 80        sigma     = torch.exp(log_sigma).clamp(min=1e-4)
 81
 82        return pi, mu, sigma, h_new, c_new

102    @staticmethod
103    def mdn_loss(
104        pi: torch.Tensor,
105        mu: torch.Tensor,
106        sigma: torch.Tensor,
107        z_next: torch.Tensor,
108    ) -> torch.Tensor:
109        """
110        Negative log-likelihood du mélange de gaussiennes (log-sum-exp numériquement stable).
111
112        log p(z_{t+1}) = logsumexp_k [ log π_k + log N(z_{t+1} | μ_k, σ_k) ]
113
114        :param pi: Logits de mélange non normalisés, shape (B, num_gaussians).
115        :type pi: torch.Tensor
116        :param mu: Moyennes des gaussiennes, shape (B, num_gaussians, latent_dim).
117        :type mu: torch.Tensor
118        :param sigma: Écarts-types des gaussiennes, shape (B, num_gaussians, latent_dim).
119        :type sigma: torch.Tensor
120        :param z_next: Prochain vecteur latent cible, shape (B, latent_dim).
121        :type z_next: torch.Tensor
122        :returns: NLL moyenne sur le batch.
123        :rtype: torch.Tensor
124        """
125        z_next = z_next.unsqueeze(1).expand_as(mu)
126        log_prob = torch.distributions.Normal(mu, sigma).log_prob(z_next).sum(dim=-1)
127        log_pi = torch.log_softmax(pi, dim=-1)
128        return -torch.logsumexp(log_pi + log_prob, dim=-1).mean()

108    ) -> torch.Tensor:
109        """
110        Negative log-likelihood du mélange de gaussiennes (log-sum-exp numériquement stable).
111
112        log p(z_{t+1}) = logsumexp_k [ log π_k + log N(z_{t+1} | μ_k, σ_k) ]
113
114        :param pi: Logits de mélange non normalisés, shape (B, num_gaussians).
115        :type pi: torch.Tensor
116        :param mu: Moyennes des gaussiennes, shape (B, num_gaussians, latent_dim).
117        :type mu: torch.Tensor
118        :param sigma: Écarts-types des gaussiennes, shape (B, num_gaussians, latent_dim).
119        :type sigma: torch.Tensor
120        :param z_next: Prochain vecteur latent cible, shape (B, latent_dim).
121        :type z_next: torch.Tensor
122        :returns: NLL moyenne sur le batch.
123        :rtype: torch.Tensor
124        """
125        z_next = z_next.unsqueeze(1).expand_as(mu)
126        log_prob = torch.distributions.Normal(mu, sigma).log_prob(z_next).sum(dim=-1)
127        log_pi = torch.log_softmax(pi, dim=-1)
128        return -torch.logsumexp(log_pi + log_prob, dim=-1).mean()
